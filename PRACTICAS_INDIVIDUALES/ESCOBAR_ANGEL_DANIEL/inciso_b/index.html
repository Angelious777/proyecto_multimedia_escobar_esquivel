B. Filtro de Suavizado

Diseño del Formulario

Vista del formulario en Windows Forms. Se utilizan dos controles PictureBox para mostrar la imagen original y la imagen procesada, facilitando la comparación de los resultados.

Código del Filtro en C#

Fragmento del código encargado de recorrer los píxeles de la imagen y aplicar el filtro. El proceso analiza cada píxel junto con los que lo rodean para generar una nueva imagen.

Carga de la Imagen Original

Ejecución de la aplicación mostrando la imagen antes de aplicar cualquier filtro. La imagen se carga y se visualiza en el primer PictureBox.

Resultado del Filtro

Imagen obtenida después de aplicar el filtro. Se puede observar un efecto de suavizado que reduce cambios bruscos entre píxeles y produce una apariencia más uniforme.

Código Fuente — Solución Escritorio (C#)

using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.Data;
using System.Drawing;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using System.Windows.Forms;

namespace Filtro_Suavizado_Practica
{
    public partial class Form1 : Form
    {
        Bitmap imagenOriginal;

        public Form1()
        {
            InitializeComponent();
        }

        // Carga dinámica de mapas de bits con extensiones filtradas
        private void btnCargar_Click(object sender, EventArgs e)
        {
            OpenFileDialog ofd = new OpenFileDialog();
            ofd.Filter = "Imágenes|*.jpg;*.png;*.bmp";

            if (ofd.ShowDialog() == DialogResult.OK)
            {
                imagenOriginal = new Bitmap(ofd.FileName);
                pictureBox1.Image = imagenOriginal;
            }
        }

        // Procesamiento Digital de Imágenes: Máscara Convolucional 3x3 de Promedio
        private void btnFiltro_Click(object sender, EventArgs e)
        {
            if (imagenOriginal == null) return;

            Bitmap resultado = new Bitmap(imagenOriginal.Width, imagenOriginal.Height);

            // Bucle espacial preservando los bordes de la imagen periférica
            for (int y = 1; y < imagenOriginal.Height - 1; y++)
            {
                for (int x = 1; x < imagenOriginal.Width - 1; x++)
                {
                    int sumaR = 0;
                    int sumaG = 0;
                    int sumaB = 0;

                    // Convolución discreta en vecindad de radio unitario (Kernel 3x3)
                    for (int j = -1; j <= 1; j++)
                    {
                        for (int i = -1; i <= 1; i++)
                        {
                            Color pixel = imagenOriginal.GetPixel(x + i, y + j);

                            sumaR += pixel.R;
                            sumaG += pixel.G;
                            sumaB += pixel.B;
                        }
                    }

                    // Cálculo aritmético del promedio de los 9 píxeles circundantes
                    int promedioR = sumaR / 9;
                    int promedioG = sumaG / 9;
                    int promedioB = sumaB / 9;

                    Color nuevoColor = Color.FromArgb(promedioR, promedioG, promedioB);
                    resultado.SetPixel(x, y, nuevoColor);
                }
            }

            pictureBox2.Image = resultado;
        }
    }
}
