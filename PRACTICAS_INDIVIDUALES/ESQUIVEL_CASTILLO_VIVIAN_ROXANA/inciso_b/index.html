Código Fuente — Solución Escritorio (C#)

using System;
using System.Drawing;
using System.Windows.Forms;

namespace filtro_suavizado
{
    public partial class Form1 : Form
    {
        Bitmap bmpOriginal;
        Bitmap bmpResultado;

        public Form1() { InitializeComponent(); }

        private void btnCargar_Click(object sender, EventArgs e)
        {
            openFileDialog1.Filter = "Archivos Imagen|*.jpg;*.png;*.jpeg";
            if (openFileDialog1.ShowDialog() == DialogResult.OK)
            {
                bmpOriginal = new Bitmap(openFileDialog1.FileName);
                picOriginal.Image = bmpOriginal;
            }
        }

        private void btnFiltrar_Click(object sender, EventArgs e)
        {
            if (bmpOriginal == null) return;

            int anchoReal = bmpOriginal.Width;
            int altoReal = bmpOriginal.Height;
            bmpResultado = new Bitmap(anchoReal, altoReal);

            for (int x = 1; x < anchoReal - 1; x++)
            {
                for (int y = 1; y < altoReal - 1; y++)
                {
                    int sumaR = 0, sumaG = 0, sumaB = 0;

                    for (int kx = -1; kx <= 1; kx++)
                    {
                        for (int ky = -1; ky <= 1; ky++)
                        {
                            Color pixelVecino = bmpOriginal.GetPixel(x + kx, y + ky);
                            sumaR += pixelVecino.R;
                            sumaG += pixelVecino.G;
                            sumaB += pixelVecino.B;
                        }
                    }

                    bmpResultado.SetPixel(x, y, Color.FromArgb(sumaR / 9, sumaG / 9, sumaB / 9));
                }
            }
            picResultado.Image = bmpResultado;
        }
    }
}
