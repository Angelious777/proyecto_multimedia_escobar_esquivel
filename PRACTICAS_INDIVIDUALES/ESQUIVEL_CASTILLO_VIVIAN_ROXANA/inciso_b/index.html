B. Filtro de Suavizado

Diseño del Formulario

Vista del formulario desarrollado en Windows Forms. Se utilizan dos controles PictureBox para mostrar la imagen original y la imagen procesada, permitiendo comparar fácilmente los resultados.

Código del Filtro

Fragmento del código en C# encargado de recorrer los píxeles de la imagen y aplicar el filtro. El proceso analiza cada píxel junto con los que lo rodean para generar una nueva imagen suavizada.

Imagen Original

Estado inicial de la aplicación después de cargar la imagen. En esta etapa todavía no se ha aplicado ningún procesamiento y se observa la imagen tal como fue seleccionada.

Resultado del Filtro

Imagen obtenida después de aplicar el filtro de promedio 3×3. Se aprecia un efecto de suavizado que reduce las diferencias bruscas entre píxeles y mejora la uniformidad de la imagen.

Código Fuente — Solución Escritorio (C#)

using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.Data;
using System.Drawing;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using System.Windows.Forms;

namespace filtro_suavizado
{
    public partial class Form1 : Form
    {
        Bitmap bmpOriginal;
        Bitmap bmpResultado;
        Bitmap bmpBordes; 
        public Form1()
        {
            InitializeComponent();
        }

        private void openFileDialog1_FileOk(object sender, CancelEventArgs e)
        {

        }

        private void btnCargar_Click(object sender, EventArgs e)
        {
            openFileDialog1.Filter = "Archivos JPG|*.jpg| Archivos PNG|*.png| Archivos JPEG|*.jpeg";
            openFileDialog1.InitialDirectory = Environment.GetFolderPath(Environment.SpecialFolder.Desktop);

            if (openFileDialog1.ShowDialog() == DialogResult.OK)
            {
                bmpOriginal = new Bitmap(openFileDialog1.FileName);
                picOriginal.Image = bmpOriginal;
            }
        }

        private void btnFiltrar_Click(object sender, EventArgs e)
        {
            if (bmpOriginal == null)
            {
                MessageBox.Show("Carga una imagen primero");
                return;
            }

            int anchoReal = bmpOriginal.Width;
            int altoReal = bmpOriginal.Height;

            bmpResultado = new Bitmap(anchoReal, altoReal);
            bmpBordes = new Bitmap(anchoReal, altoReal);
            int[,] kernelLaplace = new int[3, 3] {
                {  -1,  -1,  -1 },
                {  -1, 8,  -1 },
                {  -1,  1,  -1 }
            };

            int umbralBorde = 25;

            for (int x = 0; x < anchoReal; x++)
            {
                for (int y = 0; y < altoReal; y++)
                {
                    if (x == 0 || y == 0 || x == anchoReal - 1 || y == altoReal - 1)
                    {
                        bmpResultado.SetPixel(x, y, bmpOriginal.GetPixel(x, y));
                    }
                }
            }

            for (int x = 1; x < anchoReal - 1; x++)
            {
                for (int y = 1; y < altoReal - 1; y++)
                {
                    // 1. PASO: DETECCIÓN DE BORDE CON LAPLACE
                    int laplaceR = 0, laplaceG = 0, laplaceB = 0;

                    for (int kx = -1; kx <= 1; kx++)
                    {
                        for (int ky = -1; ky <= 1; ky++)
                        {
                            Color pixelVecino = bmpOriginal.GetPixel(x + kx, y + ky);
                            int peso = kernelLaplace[kx + 1, ky + 1];

                            laplaceR += pixelVecino.R * peso;
                            laplaceG += pixelVecino.G * peso;
                            laplaceB += pixelVecino.B * peso;
                        }
                    }

                    int magnitudBorde = (Math.Abs(laplaceR) + Math.Abs(laplaceG) + Math.Abs(laplaceB)) / 3;
                    int valorBorde = Math.Min(255, magnitudBorde);
                    bmpBordes.SetPixel(x, y, Color.FromArgb(valorBorde, valorBorde, valorBorde));

                    // 2. PASO: CONDICIONAL SEGÚN TU LICENCIADO
                    if (magnitudBorde > umbralBorde)
                    {
                        // SI ES UN BORDE -> Aplicamos el suavizado 
                        int sumaR = 0, sumaG = 0, sumaB = 0;

                        for (int kx = -1; kx <= 1; kx++)
                        {
                            for (int ky = -1; ky <= 1; ky++)
                            {
                                Color pixelVecino = bmpOriginal.GetPixel(x + kx, y + ky);
                                sumaR += pixelVecino.R;
                                sumaG += pixelVecino.G;
                                sumaB += pixelVecino.B;
                            }
                        }

                        int promR = sumaR / 9;
                        int promG = sumaG / 9;
                        int promB = sumaB / 9;

                        bmpResultado.SetPixel(x, y, Color.FromArgb(promR, promG, promB));
                    }
                    else
                    {
                        // NO ES UN BORDE -> Se queda el color original de la foto sin alterar
                        bmpResultado.SetPixel(x, y, bmpOriginal.GetPixel(x, y));
                    }
                }
            }

            picResultado.Image = bmpResultado;
            pctBorde.Image = bmpBordes;
            MessageBox.Show("Filtro de suavizado por Laplace aplicado con éxito");
        }

        private void picOriginal_Click(object sender, EventArgs e)
        {

        }
    }
