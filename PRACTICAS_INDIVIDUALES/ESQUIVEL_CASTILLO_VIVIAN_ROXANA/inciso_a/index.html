Implementación del Lógica del Backend (C#)

Dado que este módulo requiere una conexión activa a un servidor de base de datos relacional local (SQL Server), la ejecución en tiempo real mediante botones web directos está deshabilitada por restricciones de arquitectura de red del navegador. A continuación se expone el bloque de código C# implementado para este fin:

Form1.cs — ReconocimientoZonas C# 
using System;
using System.Data;
using System.Drawing;
using System.Windows.Forms;
using System.Data.SqlClient;

namespace reconocimiento_zonas
{
    public partial class Form1 : Form
    {
        Bitmap bmp;
        int lastX, lastY;

        // Captura el color RGB del píxel escalado en el PictureBox
        private void pictureBox1_MouseClick(object sender, MouseEventArgs e)
        {
            if (bmp == null) return;
            Point p = GetImageCoordinates(pictureBox1, e);
            lastX = p.X; lastY = p.Y;

            if (lastX < 0 || lastY < 0) return;
            Color c = bmp.GetPixel(lastX, lastY);

            txtR.Text = c.R.ToString();
            txtG.Text = c.G.ToString();
            txtB.Text = c.B.ToString();
        }

        // Guarda el vector de color y su etiqueta en SQL Server
        private void btnGuardar_Click(object sender, EventArgs e)
        {
            using (SqlConnection con = new SqlConnection("server=(local); user=multimedia; pwd=1234; database=bd_Texturas"))
            {
                string query = "INSERT INTO reconocimiento_puntos(nombre_punto, tipo_zona, R, G, B) VALUES ('"
                             + txtNombre.Text + "','" + txtTipoZona.Text + "'," + txtR.Text + "," + txtG.Text + "," + txtB.Text + ")";
                SqlCommand cmd = new SqlCommand(query, con);
                con.Open();
                cmd.ExecuteNonQuery();
            }

            // Actualiza de inmediato el DataGridView de la interfaz
            SqlDataAdapter ada = new SqlDataAdapter("SELECT * FROM reconocimiento_puntos", "server=(local); user=multimedia; pwd=1234; database=bd_Texturas");
            DataSet ds = new DataSet();
            ada.Fill(ds);
            dataGridView1.DataSource = ds.Tables[0];
        }

        // Algoritmo Clasificador por Distancia Absoluta (Manhattan)
        private void btnReconocer_Click(object sender, EventArgs e)
        {
            if (bmp == null || lastX < 0) return;
            Color c = bmp.GetPixel(lastX, lastY);

            using (SqlConnection con = new SqlConnection("server=(local); user=multimedia; pwd=1234; database=bd_Texturas"))
            {
                con.Open();
                SqlCommand cmd = new SqlCommand("SELECT * FROM reconocimiento_puntos", con);
                SqlDataReader dr = cmd.ExecuteReader();

                string zona = "";
                int mejor = int.MaxValue;

                while (dr.Read())
                {
                    int r = Convert.ToInt32(dr["R"]);
                    int g = Convert.ToInt32(dr["G"]);
                    int b = Convert.ToInt32(dr["B"]);

                    // Distancia matemática de bloques (Manhattan)
                    int dist = Math.Abs(c.R - r) + Math.Abs(c.G - g) + Math.Abs(c.B - b);

                    if (dist < mejor)
                    {
                        mejor = dist;
                        zona = dr["tipo_zona"].ToString();
                    }
                }
                lblResultado.Text = zona;
                MessageBox.Show("Zona detectada: " + zona);
            }
        }

        // Ajusta las coordenadas del clic al tamaño real de la imagen cargada
        private Point GetImageCoordinates(PictureBox pb, MouseEventArgs e)
        {
            if (pb.Image == null) return new Point(-1, -1);
            return new Point((int)(e.X * (pb.Image.Width / (float)pb.Width)), (int)(e.Y * (pb.Image.Height / (float)pb.Height)));
        }
    }
}
